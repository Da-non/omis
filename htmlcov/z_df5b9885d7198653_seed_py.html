17def _records(dep: datetime, status: S, scale: float = 1.0): 

18 if status in (S.DELAYED, S.NOT_DELIVERED): 

19 seq = STATUS_FLOW[:4] + [status] 

20 else: 

21 seq = STATUS_FLOW[:STATUS_FLOW.index(status) + 1] 

22 return [TrackingRecord(status=s, timestamp=min(dep + timedelta(hours=OFFSET_H[s] * scale), datetime.now()), 

23 description=s.value) for s in seq]
